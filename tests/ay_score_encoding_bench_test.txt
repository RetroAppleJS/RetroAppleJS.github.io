'use strict';
const test=require('node:test'),assert=require('node:assert/strict');
const path=require('node:path');
const fs=require('node:fs');
const B=require('../tools/AY_SCORE_ENCODING_BENCH_core.js');
const opts={musicBudget:20,graphicsPercent:50,reservePercent:10,graphicsProfile:'steady',chunkSize:256,cyclesPerWrite:50};
const modes=['E0','E1','E2','E3','E4'];
function roundtrip(score,codecs=modes){for(const c of codecs){const e=B.encode(c,score,{chunkSize:256}),d=B.decode(e.binary);assert.deepEqual(d.events,score.events,`codec ${c}`);assert.equal(e.roundTrip.equal,true);assert.equal(e.binary.length,e.bytes);}}

test('public API and stable format identity',()=>{assert.equal(B.VERSION,1);assert.equal(B.CODECS.length,5);assert.match(B.SCHEMA,/score-encoding-bench/);});
test('six deterministic scenarios produce note and AY activity',()=>{for(const scenario of ['chord','arpeggio','bass','mixed','ambient','action']){const a=B.generate({scenario,seconds:3,seed:42}),b=B.generate({scenario,seconds:3,seed:42});assert.deepEqual(a,b);assert.ok(a.actions.length>0);assert.ok(a.events.length>0);assert.ok(a.events.every(e=>e.cycle>=0&&e.cycle<=a.durationCycles));}});
test('changing the seed changes mixed score',()=>{assert.notDeepEqual(B.generate({seconds:3,seed:1}).events,B.generate({seconds:3,seed:2}).events);});
test('E0, E1, E2, E3, E4 exact roundtrips on five scenarios',()=>{for(const scenario of ['chord','arpeggio','bass','mixed','ambient'])roundtrip(B.generate({scenario,seconds:3}));});
test('E3 rejects free AY writes; E4 supports mixed VM/direct',()=>{const s=B.generate({scenario:'action',seconds:4});assert.throws(()=>B.encode('E3',s),/direct writes/);roundtrip(s,['E0','E1','E2','E4']);assert.equal(B.encode('E4',s).stats.hybridMode,'VM with direct escapes');});
test('sustained notes strongly compress with E3',()=>{const s=B.generate({scenario:'chord',seconds:5});const raw=B.encode('E0',s),vm=B.encode('E3',s);assert.ok(vm.bytes<raw.bytes/20);});
test('pattern backreferences occur and reduce a repeated DAC workload',()=>{const s=B.generate({scenario:'bass',seconds:5});const p=B.encode('E2',s,{chunkSize:512});assert.ok(p.stats.backrefs>0);assert.ok(p.stats.copied>0);assert.ok(p.bytes<B.encode('E1',s).bytes);});
test('E2 block boundaries 64, 128, 256, 512, 1024, 2048',()=>{const s=B.generate({scenario:'mixed',seconds:3});for(const chunkSize of [64,128,256,512,1024,2048])roundtrip(s,['E2']);});
test('E2 and delta handle zero-time collocated events',()=>{const s=B.fromEventJson({cpuHz:1023000,ayHz:1021800,durationCycles:50000,events:[{cycle:0,chip:0,reg:0,value:255},{cycle:0,chip:0,reg:1,value:15},{cycle:0,chip:1,reg:13,value:10},{cycle:29999,chip:0,reg:8,value:0},{cycle:30000,chip:0,reg:8,value:12}]});roundtrip(s,['E0','E1','E2','E4']);assert.throws(()=>B.encode('E3',s),/requires supported notes/);});
test('E4 handles event-only imports via LZ fallback',()=>{const s=B.fromEventJson({events:[{cycle:10,chip:0,reg:8,value:3},{cycle:20,chip:1,reg:8,value:7}]});const e=B.encode('E4',s);assert.equal(e.stats.hybridMode,'LZ event fallback');roundtrip(s,['E4']);});
test('empty event stream remains valid',()=>{const s=B.fromEventJson({events:[],durationCycles:1023000});roundtrip(s,['E0','E1','E2','E4']);});
test('midi, parameters, bounds and malformed input fail clearly',()=>{assert.throws(()=>B.generate({seconds:91}),/duration/);assert.throws(()=>B.generate({scenario:'a'}),/scenario/);assert.throws(()=>B.fromEventJson({events:[{cycle:-1,chip:0,reg:8,value:3}]}),/cycle/);assert.throws(()=>B.fromEventJson({events:[{cycle:0,chip:2,reg:8,value:3}]}),/chip/);});
test('decode rejects tampered header, checksum, length and event count',()=>{const s=B.generate({scenario:'ambient',seconds:3}),original=B.encode('E1',s).binary;const p=Uint8Array.from(original);p[0]=0;assert.throws(()=>B.decode(p),/not AYSB/);const q=Uint8Array.from(original);q[q.length-1]^=1;assert.throws(()=>B.decode(q),/checksum/);const short=original.slice(0,-1);assert.throws(()=>B.decode(short),/payload length/);const r=Uint8Array.from(original);r[20]=0xff;r[21]=0xff;r[22]=0xff;r[23]=0xff;assert.throws(()=>B.decode(r),/safety limits/);});
test('binary reader rejects unsupported version or flag',()=>{const s=B.generate({scenario:'ambient',seconds:2});const b=B.encode('E0',s).binary;const ver=Uint8Array.from(b);ver[4]=2;assert.throws(()=>B.decode(ver),/version/);const flag=Uint8Array.from(b);flag[6]=1;assert.throws(()=>B.decode(flag),/flags/);});
test('import Waveform Bench uses correct nanosecond-to-cycle rounding',()=>{const input={schema:'retroapplejs.ay.waveform-bench.v1.4.survey',results:[{spec:{fixture:'square',duration:.5,clock:1021800,events:[{tick:0,chip:0,reg:8,value:0},{tick:1000000,chip:0,reg:8,value:15},{tick:1000100,chip:0,reg:8,value:5}]}}]};const s=B.fromWaveformSurvey(input,{cpuHz:1023000});assert.deepEqual(s.events.map(x=>x.cycle),[0,1023,1023]);roundtrip(s,['E0','E1','E2','E4']);});
test('import selected fixture by result index',()=>{const data={results:[{spec:{clock:1021800,duration:1,fixture:'a',events:[]}},{spec:{clock:1021800,duration:2,fixture:'b',events:[{tick:42,chip:0,reg:8,value:7}]}}]};const s=B.fromWaveformSurvey(data,{resultIndex:1});assert.equal(s.importInfo.fixture,'b');assert.equal(s.events.length,1);});
test('event sorting stable at identical timestamp',()=>{const s=B.fromEventJson({events:[{cycle:1,chip:1,reg:8,value:2},{cycle:0,chip:0,reg:8,value:3},{cycle:1,chip:1,reg:8,value:4}]});assert.deepEqual(s.events.map(e=>e.value),[3,2,4]);roundtrip(s,['E0','E1','E2','E4']);});
test('score encodings preserve two-chip AY coordinates',()=>{const s=B.fromEventJson({durationCycles:1000,events:[{cycle:0,chip:1,reg:15,value:255},{cycle:9,chip:0,reg:0,value:23},{cycle:99,chip:1,reg:8,value:16}]});roundtrip(s,['E0','E1','E2','E4']);});
test('schedule is deterministic and higher CPU caps help',()=>{const s=B.generate({scenario:'bass',seconds:3}),e=B.encode('E2',s);const a=B.simulateSchedule(s,e,{...opts,musicBudget:5}),b=B.simulateSchedule(s,e,{...opts,musicBudget:40});assert.deepEqual(a,B.simulateSchedule(s,e,{...opts,musicBudget:5}));assert.ok(a.maxLatenessUs>=b.maxLatenessUs);assert.ok(a.missedDeadlines>=b.missedDeadlines);});
test('increasing VIA cycles raises demand and misses',()=>{const s=B.generate({scenario:'chord',seconds:3}),e=B.encode('E2',s);const a=B.simulateSchedule(s,e,{...opts,cyclesPerWrite:20}),b=B.simulateSchedule(s,e,{...opts,cyclesPerWrite:120});assert.ok(b.estimatedTotalCycles>a.estimatedTotalCycles);assert.ok(b.missedDeadlines>=a.missedDeadlines);});
test('graphics action profile constrains available frame capacity',()=>{const s=B.generate({scenario:'chord',seconds:4}),e=B.encode('E2',s);const a=B.simulateSchedule(s,e,{...opts,graphicsProfile:'steady',graphicsPercent:60,musicBudget:80}),b=B.simulateSchedule(s,e,{...opts,graphicsProfile:'action',graphicsPercent:60,musicBudget:80});assert.ok(Math.min(...b.frameTimeline.map(f=>f.availablePercent))<Math.min(...a.frameTimeline.map(f=>f.availablePercent)));});
test('memory includes score + decoder + state in resident mode',()=>{const s=B.generate({scenario:'chord',seconds:4}),r=B.encode('E3',s),mem=B.memoryModel(s,r,opts);assert.equal(mem.fullyResidentBytes,mem.scoreBytes+mem.playerCodeBytes+mem.instrumentBytes+mem.workingRamBytes);assert.equal(mem.streamingResidentBytes,mem.playerCodeBytes+mem.instrumentBytes+mem.workingRamBytes+2*512);});
test('benchmark reports all codecs and Pareto labels',()=>{const s=B.generate({scenario:'mixed',seconds:3}),r=B.benchmark(s,opts);assert.equal(r.results.length,5);assert.ok(r.results.every(x=>x.status==='ok'));assert.ok(r.results.some(x=>x.pareto));const report=B.exportReport(r);assert.ok(!report.results[0].encoded);assert.ok(report.results[0].bytes>0);assert.match(B.csv(r),/scoreBytes/);});
test('hex preview truncates safely',()=>{const bytes=Uint8Array.from({length:100},(_,i)=>i);const h=B.hex(bytes,32);assert.match(h,/000000:/);assert.match(h,/68 more bytes/);});
test('roundtrip against genuine v1.4 generated events from upload',()=>{const filename='/mnt/data/ay-buzzer-phase-lock-v1.4.json';if(!fs.existsSync(filename))return;const data=JSON.parse(fs.readFileSync(filename,'utf8')),s=B.fromWaveformSurvey(data,{resultIndex:0,cpuHz:1023000});assert.ok(s.events.length>8);roundtrip(s,['E0','E1','E2','E4']);});
test('score importer roundtrips high-density Waveform Bench DAC events',()=>{const filename='/mnt/data/ay-triangle-budget-v1.4.json';if(!fs.existsSync(filename))return;const data=JSON.parse(fs.readFileSync(filename,'utf8'));const ix=data.results.findIndex(r=>r.spec.fixture==='triangle-dac-sparse'&&r.spec.hz===220);assert.ok(ix>=0);const s=B.fromWaveformSurvey(data,{resultIndex:ix});assert.ok(s.events.length>100);roundtrip(s,['E0','E1','E2','E4']);});

test('stream buffers are independent of pattern token block size',()=>{const s=B.generate({scenario:'bass',seconds:2}),e=B.encode('E2',s,{chunkSize:128});const a=B.memoryModel(s,e,{...opts,streamBufferBytes:256,chunkSize:128}),b=B.memoryModel(s,e,{...opts,streamBufferBytes:1024,chunkSize:2048});assert.equal(a.streamBuffersBytes,512);assert.equal(b.streamBuffersBytes,2048);assert.equal(a.workingRamBytes,b.workingRamBytes);});

test('no available CPU reports scheduling failure without mislabeling codec',()=>{const s=B.generate({scenario:'ambient',seconds:2}),r=B.benchmark(s,{musicBudget:40,graphicsPercent:95,reservePercent:10,graphicsProfile:'steady'});assert.ok(r.results.every(x=>x.status==='ok'));assert.ok(r.results.every(x=>x.cycles.capacityExhausted===true&&x.cycles.missedDeadlines===s.events.length));});

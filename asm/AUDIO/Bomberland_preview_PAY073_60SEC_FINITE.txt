; {"codeVariants":338,"stateGraphNodes":2836} advisory evidence caps; no SID/data-read drops accepted
; VIA0=$C400 VIA1=$C480 physical slot=4; source mode=faithful; end=stop
; ENTRY: PAY_INIT=$6000 PAY_TICK=$6003 PAY_STOP=$6006 PAY_IRQ_INSTALL=$6009 PAY_IRQ_REMOVE=$600C
; PEEK diagnostics: FRAME_LO=$B380, FRAME_HI=$B381 IRQ_ACTIVE=$B38F IRQ_COUNT_LO=$B393
; No source-level song loop proven. Repeat window is artificial if selected.
; Monitor IRQ $03FE uses $45 ROM scratch. VIA1 T1 must be exclusive. IRQ install executes CLI.
; This is a bounded 6502 PSID trace, not exact SID envelope/filter emulation.
; PAY-00.7.3 readable score map accompanies this assembly.

PAY_LOAD_HEADER:
  ; Read next (deltaLo,deltaHi,count), accumulate absolute target frame.
  JSR PAY_GET_BYTE
  STA PAY_DELTA_LO
  JSR PAY_GET_BYTE
  STA PAY_DELTA_HI
  CLC 
  LDA NEXT_FRAME_LO
  ADC PAY_DELTA_LO
  STA NEXT_FRAME_LO
  LDA NEXT_FRAME_HI
  ADC PAY_DELTA_HI
  STA NEXT_FRAME_HI
  JSR PAY_GET_BYTE
  STA PAY_UPDATE_COUNT
  RTS

  BNE PAY_CHECK_EVENT
  LDA FRAME_LO
  CMP #$B8
  BNE PAY_CHECK_EVENT
  JMP PAY_AT_END
PAY_CHECK_EVENT:
  LDA EVENTS_LEFT_LO
  ORA EVENTS_LEFT_HI
  BEQ PAY_ADVANCE
  LDA FRAME_HI
  CMP NEXT_FRAME_HI
  BNE PAY_ADVANCE
  LDA FRAME_LO
  CMP NEXT_FRAME_LO
  BNE PAY_ADVANCE
  JSR PAY_PROCESS_EVENT
PAY_ADVANCE:
  INC FRAME_LO
  BNE PAY_RETURN
  INC FRAME_HI
PAY_RETURN:

  TAY 
  PLA 
  TAX 
  PLA 
  RTS 
PAY_AT_END:
  LDA #$01
  STA PAY_DONE
  JSR PAY_MUTE
  JMP PAY_RETURN
PAY_PROCESS_EVENT:
  ; Apply all AY writes for the current source frame.
PAY_UPDATE_LOOP:
  LDA PAY_UPDATE_COUNT
  BEQ PAY_EVENT_FINISH
  JSR PAY_GET_BYTE
  STA PAY_REG_TMP
  JSR PAY_GET_BYTE
  LDX PAY_REG_TMP
  JSR PAY_AY_WRITE
  DEC PAY_UPDATE_COUNT
  JMP PAY_UPDATE_LOOP
PAY_EVENT_FINISH:
  LDA EVENTS_LEFT_LO
  BNE PAY_LEFT_LOW
  DEC EVENTS_LEFT_HI
PAY_LEFT_LOW:
  DEC EVENTS_LEFT_LO
  LDA EVENTS_LEFT_LO
  ORA EVENTS_LEFT_HI
  BEQ PAY_EVENT_FINISHED
  JSR PAY_LOAD_HEADER
  RTS 
PAY_EVENT_FINISHED:
  LDA #$FF
  STA NEXT_FRAME_LO
  LDA #$FF
  STA NEXT_FRAME_HI
  RTS
